Result

Delivered an optimized Dashboard.tsx (full file below; original kept as Dashboard.before.tsx in the scratch folder). It type-checks under strict TS, and I benchmarked before vs after with the same harness (jsdom + React Profiler, 2,000 appointments, 10 therapists, mocked 100 ms API, chart.js mocked with a construct counter). What was slow and what I changed 1. The 1 s clock lived in the top-level component, so every second re-rendered all 2,000 rows (each calling moment). Moved it into a memoized <Clock/>. 5 ticks: ~1,400-2,260 ms of render work before, ~0.4 ms after. 2. The interval effect had no dependency array, so it was torn down and re-created after every render (7 intervals registered during the run, now 1). 3. The filter + sort + group ran on every render, and the sort parsed two moment objects per comparison. Now one useMemo for the sort (ISO-8601 strings compare lexically, no parsing), one for filter+group (a Map), lodash dropped. Search input uses useDeferredValue so typing stays responsive. 4. Rows: key={i} replaced by key={a.id}; Row is React.memo; onCancel is a stable useCallback taking an id (functional setState), therapist is looked up in a Map instead of therapists.find per row. Cancelling one appointment: ~380-400 ms before, ~6-9 ms after (only the removed row's section list changes; the other 1,998 rows are memo-skipped). 5. StatsChart created a new chart.js instance after every render and never destroyed it (17 charts constructed during the run, 0 destroyed; a real canvas would throw "Canvas is already in use"). Now created once, updated in place when data changes, destroyed on unmount. 6. Three awaited fetches in a row (a waterfall) became Promise.all, with AbortController cleanup and an error message. Data ready at ~320 ms before, ~115-125 ms after (3 x 100 ms latency vs 1 x). 7. moment and lodash removed (Intl.DateTimeFormat and a Map instead), and chart.js is imported dynamically. Initial JS for the component: 345 KB (119 KB gzip) before -> 2.5 KB (1.3 KB gzip) after; the 205 KB chart.js chunk loads only when stats arrive. Caveats: the first benchmark runs were noisy because the machine was shared with other jobs; numbers quoted are the steadier runs (the full output of all 3 runs is in the steps). Render times are React Profiler actualDuration in a dev build under jsdom, so absolute numbers are not browser numbers; the ratios and commit/instance counts are the reliable part. Typing causes 2 commits per keystroke after (urgent input commit + deferred list commit), which is how useDeferredValue works; total render time is still ~25x lower. Behavior preserved: same markup, same cancel semantics. I did not test against a real chart.js canvas (mocked).

Dashboard.tsx (optimized)
// Tidewater clinic dashboard – optimized.
import * as React from class="s">"react"
import type { Chart as ChartType } from class="s">"chart.js"

type Appointment = { id: string; patient: string; start: string; therapistId: string; status: class="s">"booked" | class="s">"cancelled" | class="s">"done" }
type Therapist = { id: string; name: string }
type Stats = { labels: string[]; values: number[] }

// One shared formatter instead of building a moment object per row per render.
const hhmm = new Intl.DateTimeFormat(class="s">"en-GB", { hour: class="s">"2-digit", minute: class="s">"2-digit", hour12: false })
const hhmmss = new Intl.DateTimeFormat(class="s">"en-GB", { hour: class="s">"2-digit", minute: class="s">"2-digit", second: class="s">"2-digit", hour12: false })

const getJson = <T,>(url: string, signal: AbortSignal): Promise<T> =>
  fetch(url, { signal }).then((r) => {
    if (!r.ok) throw new Error(class="s">`${url} -> ${r.status}`)
    return r.json()
  })

export default function Dashboard({ clinicId }: { clinicId: string }) {
  const [appointments, setAppointments] = React.useState<Appointment[]>([])
  const [therapists, setTherapists] = React.useState<Therapist[]>([])
  const [stats, setStats] = React.useState<Stats | null>(null)
  const [error, setError] = React.useState<string | null>(null)
  const [search, setSearch] = React.useState(class="s">"")
  // Typing stays instant; the heavy filter/group runs on a deferred value.
  const deferredSearch = React.useDeferredValue(search)

  // The three requests are independent: fetch them in parallel, cancel on unmount / clinic change.
  React.useEffect(() => {
    const ctrl = new AbortController()
    const base = class="s">`/api/clinics/${clinicId}`
    Promise.all([
      getJson<Appointment[]>(class="s">`${base}/appointments`, ctrl.signal),
      getJson<Therapist[]>(class="s">`${base}/therapists`, ctrl.signal),
      getJson<Stats>(class="s">`${base}/stats`, ctrl.signal),
    ])
      .then(([a, t, s]) => {
        setAppointments(a)
        setTherapists(t)
        setStats(s)
      })
      .catch((e) => {
        if (e.name !== class="s">"AbortError") setError(String(e.message ?? e))
      })
    return () => ctrl.abort()
  }, [clinicId])

  // ISO-8601 strings sort correctly as strings: no date parsing in the comparator.
  const sorted = React.useMemo(() => [...appointments].sort((a, b) => (a.start < b.start ? -1 : a.start > b.start ? 1 : 0)), [appointments])

  const byTherapist = React.useMemo(() => {
    const q = deferredSearch.toLowerCase()
    const groups = new Map<string, Appointment[]>()
    for (const a of sorted) {
      if (q && !a.patient.toLowerCase().includes(q)) continue
      const list = groups.get(a.therapistId)
      if (list) list.push(a)
      else groups.set(a.therapistId, [a])
    }
    return groups
  }, [sorted, deferredSearch])

  const therapistById = React.useMemo(() => new Map(therapists.map((t) => [t.id, t])), [therapists])

  // Stable identity so memoized rows are not re-rendered by unrelated state.
  const cancel = React.useCallback((id: string) => setAppointments((prev) => prev.filter((x) => x.id !== id)), [])

  return (
    <div>
      <Clock />
      <input value={search} onChange={(e) => setSearch(e.target.value)} placeholder=class="s">"Search patients" />
      {error && <p role=class="s">"alert">Could not load the dashboard: {error}</p>}
      {stats && <StatsChart data={stats} />}
      {therapists.map((t) => (
        <section key={t.id}>
          <h2>{t.name}</h2>
          {(byTherapist.get(t.id) ?? []).map((a) => (
            <Row key={a.id} appointment={a} therapist={therapistById.get(a.therapistId)!} onCancel={cancel} />
          ))}
        </section>
      ))}
    </div>
  )
}

// The 1s tick lives here, so only this <h1> re-renders every second, not the whole dashboard.
const Clock = React.memo(function Clock() {
  const [now, setNow] = React.useState(Date.now)
  React.useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), 1000)
    return () => clearInterval(id)
  }, [])
  return <h1>Today · {hhmmss.format(now)}</h1>
})

const Row = React.memo(function Row({ appointment, therapist, onCancel }: { appointment: Appointment; therapist: Therapist; onCancel: (id: string) => void }) {
  return (
    <div style={rowStyle}>
      <span>{hhmm.format(new Date(appointment.start))}</span>
      <span>{appointment.patient}</span>
      <span>{therapist.name}</span>
      <button onClick={() => onCancel(appointment.id)}>Cancel</button>
    </div>
  )
})
const rowStyle = { display: class="s">"flex", gap: 8 } as const

function StatsChart({ data }: { data: Stats }) {
  const ref = React.useRef<HTMLCanvasElement>(null)
  const chart = React.useRef<ChartType | null>(null)
  const latest = React.useRef(data) // data may change while chart.js is still loading
  latest.current = data

  // Create once (chart.js is loaded on demand, so it is not in the initial bundle); destroy on unmount.
  React.useEffect(() => {
    let cancelled = false
    import(class="s">"chart.js/auto").then(({ Chart }) => {
      if (cancelled || !ref.current) return
      chart.current = new Chart(ref.current, { type: class="s">"line", data: { labels: latest.current.labels, datasets: [{ data: latest.current.values }] } })
    })
    return () => {
      cancelled = true
      chart.current?.destroy()
      chart.current = null
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [])

  // Later data changes update the existing chart instead of constructing a new one.
  React.useEffect(() => {
    const c = chart.current
    if (!c) return
    c.data.labels = data.labels
    c.data.datasets[0].data = data.values
    c.update()
  }, [data])

  return <canvas ref={ref} />
}
