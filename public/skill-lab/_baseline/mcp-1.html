Result

Built an MCP server for the Tidewater appointments API (file server.mjs below, ~120 lines, @modelcontextprotocol/sdk 1.32.1 + zod, stdio transport) from fixtures/mcp/openapi.json, with the API mocked in memory (2 therapists, 3 patients, 2 seed appointments). It exposes one tool per OpenAPI operation: list_appointments, book_appointment, cancel_appointment, list_therapists, search_patients, get_availability. Decisions: tool names are snake_case verbs (not raw HTTP paths); zod schemas mirror the OpenAPI parameters (enum for status, datetime for start, default 45 for durationMin and true for notifyPatient); tool annotations are set (reads are readOnlyHint, cancel is destructiveHint); business errors (unknown ids, double booking, cancelling twice) come back as isError results with an actionable message that names the next tool to call, while malformed input is rejected by schema validation (JSON-RPC -32602); logging goes to stderr only because stdout is the protocol channel. The mock enforces basic rules so the calls mean something: no overlapping bookings per therapist, availability derived from the bookings (hourly 09:00-17:00 UTC slots), phone search by digits. Verified by running it for real: client.mjs spawns the server over stdio with the SDK client, lists the tools, and makes 7 calls (search, availability, a booking, a conflicting booking that returns isError, list, cancel, an invalid-input call). I also sent raw JSON-RPC (initialize + tools/call) through a pipe to confirm it works without the SDK client. Not done: HTTP transport, auth, pagination, wiring to Claude Desktop config, and the real API at api.tidewater.example (mocked as asked). Not generated automatically from the OpenAPI file; the OpenAPI has no response schemas, so outputSchema is a loose wrapper object.

import { StdioServerTransport } from class="s">"@modelcontextprotocol/sdk/server/stdio.js"
import { z } from class="s">"zod"

// ---- in-memory "API" -------------------------------------------------------------
const therapists = [
  { id: class="s">"t1", name: class="s">"Dr. Okafor", speciality: class="s">"sports injury" },
  { id: class="s">"t2", name: class="s">"Dr. Lindqvist", speciality: class="s">"neurological rehab" },
]
const patients = [
  { id: class="s">"p1", name: class="s">"Ada Lovelace", phone: class="s">"+44 20 7946 0001" },
  { id: class="s">"p2", name: class="s">"Grace Hopper", phone: class="s">"+1 202 555 0143" },
  { id: class="s">"p3", name: class="s">"Alan Turing", phone: class="s">"+44 20 7946 0999" },
]
let nextId = 2
const appointments = [
  { id: class="s">"a1", patientId: class="s">"p1", therapistId: class="s">"t1", start: class="s">"2026-10-27T09:00:00.000Z", durationMin: 45, status: class="s">"booked" },
  { id: class="s">"a2", patientId: class="s">"p2", therapistId: class="s">"t2", start: class="s">"2026-10-27T10:00:00.000Z", durationMin: 45, status: class="s">"booked" },
]
const MIN = 60_000
const overlaps = (a, startMs, durMin) => {
  const s = Date.parse(a.start)
  return a.status === class="s">"booked" && startMs < s + a.durationMin * MIN && s < startMs + durMin * MIN
}
class ApiError extends Error {}

const api = {
  listAppointments: ({ date, therapistId, status }) =>
    appointments.filter((a) => (!date || a.start.startsWith(date)) && (!therapistId || a.therapistId === therapistId) && (!status || a.status === status)),
  bookAppointment: ({ patientId, therapistId, start, durationMin = 45 }) => {
    if (!patients.some((p) => p.id === patientId)) throw new ApiError(class="s">`Unknown patientId "${patientId}". Use search_patients to find one.`)
    if (!therapists.some((t) => t.id === therapistId)) throw new ApiError(class="s">`Unknown therapistId "${therapistId}". Use list_therapists to see valid ids.`)
    const ms = Date.parse(start)
    if (Number.isNaN(ms)) throw new ApiError(class="s">`"${start}" is not an ISO date-time.`)
    const clash = appointments.find((a) => a.therapistId === therapistId && overlaps(a, ms, durationMin))
    if (clash) throw new ApiError(class="s">`${therapistId} is already booked at that time (appointment ${clash.id}). Use get_availability for free slots.`)
    const appt = { id: class="s">"a" + ++nextId, patientId, therapistId, start: new Date(ms).toISOString(), durationMin, status: class="s">"booked" }
    appointments.push(appt)
    return appt
  },
  cancelAppointment: ({ id, reason, notifyPatient = true }) => {
    const a = appointments.find((x) => x.id === id)
    if (!a) throw new ApiError(class="s">`No appointment with id "${id}".`)
    if (a.status !== class="s">"booked") throw new ApiError(class="s">`Appointment ${id} is already ${a.status}.`)
    a.status = class="s">"cancelled"
    return { ...a, cancelReason: reason ?? null, patientNotified: notifyPatient }
  },
  listTherapists: () => therapists,
  searchPatients: ({ q }) => {
    const n = q.toLowerCase(), digits = q.replace(/\D/g, class="s">"")
    return patients.filter((p) => p.name.toLowerCase().includes(n) || (digits.length >= 3 && p.phone.replace(/\D/g, class="s">"").includes(digits)))
  },
  getAvailability: ({ therapistId, date }) => {
    if (!therapists.some((t) => t.id === therapistId)) throw new ApiError(class="s">`Unknown therapistId "${therapistId}".`)
    const slots = []
    for (let h = 9; h < 17; h++) {
      const start = class="s">`${date}T${String(h).padStart(2, "0")}:00:00.000Z`
      if (!appointments.some((a) => a.therapistId === therapistId && overlaps(a, Date.parse(start), 45))) slots.push(start)
    }
    return { therapistId, date, openSlots: slots }
  },
}

// ---- MCP server --------------------------------------------------------------------
const server = new McpServer({ name: class="s">"tidewater-appointments", version: class="s">"1.0.0" })
const run = (fn) => async (args) => {
  try {
    const data = fn(args)
    return { content: [{ type: class="s">"text", text: JSON.stringify(data, null, 2) }], structuredContent: { result: data } }
  } catch (e) {
    if (e instanceof ApiError) return { isError: true, content: [{ type: class="s">"text", text: e.message }] }
    throw e
  }
}
const date = z.string().regex(/^\d{4}-\d{2}-\d{2}$/, class="s">"YYYY-MM-DD")
const out = { result: z.any() }
const RO = { readOnlyHint: true, openWorldHint: false }

server.registerTool(class="s">"list_appointments", {
  description: class="s">"List appointments, optionally filtered by date, therapist or status.",
  inputSchema: { date: date.optional().describe(class="s">"Only appointments on this day"), therapistId: z.string().optional(), status: z.enum([class="s">"booked", class="s">"cancelled", class="s">"done"]).optional() },
  outputSchema: out, annotations: { title: class="s">"List appointments", ...RO },
}, run(api.listAppointments))

server.registerTool(class="s">"book_appointment", {
  description: class="s">"Book an appointment for a patient with a therapist. Fails if the therapist is already booked in that window.",
  inputSchema: { patientId: z.string(), therapistId: z.string(), start: z.string().datetime().describe(class="s">"ISO 8601 date-time, e.g. 2026-10-28T09:00:00Z"), durationMin: z.number().int().positive().default(45).optional() },
  outputSchema: out, annotations: { title: class="s">"Book appointment", readOnlyHint: false, destructiveHint: false, idempotentHint: false, openWorldHint: false },
}, run(api.bookAppointment))

server.registerTool(class="s">"cancel_appointment", {
  description: class="s">"Cancel a booked appointment by id, optionally with a reason. Irreversible.",
  inputSchema: { id: z.string(), reason: z.string().optional(), notifyPatient: z.boolean().default(true).optional() },
  outputSchema: out, annotations: { title: class="s">"Cancel appointment", readOnlyHint: false, destructiveHint: true, idempotentHint: true, openWorldHint: false },
}, run(api.cancelAppointment))

server.registerTool(class="s">"list_therapists", { description: class="s">"List all therapists.", inputSchema: {}, outputSchema: out, annotations: { title: class="s">"List therapists", ...RO } }, run(api.listTherapists))

server.registerTool(class="s">"search_patients", {
  description: class="s">"Search patients by name or phone number.",
  inputSchema: { q: z.string().min(1).describe(class="s">"Name fragment or phone digits") },
  outputSchema: out, annotations: { title: class="s">"Search patients", ...RO },
}, run(api.searchPatients))

server.registerTool(class="s">"get_availability", {
  description: class="s">"Open hourly slots for a therapist on a given day.",
  inputSchema: { therapistId: z.string(), date },
  outputSchema: out, annotations: { title: class="s">"Get availability", ...RO },
}, run(api.getAvailability))
