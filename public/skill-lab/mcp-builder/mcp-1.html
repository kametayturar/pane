// stdio: local, single client. Never log to stdout - it carries the protocol.
async function runStdio(): Promise<void> {
  await buildServer().connect(new StdioServerTransport());
  console.error(class="s">`${SERVER_NAME} running on stdio`);
}

// Streamable HTTP, stateless JSON: a fresh server + transport per request, bound to 127.0.0.1.
async function runHttp(port: number): Promise<void> {
  const http = createServer(async (req, res) => {
    if (req.url !== class="s">"/mcp" || req.method !== class="s">"POST") {
      res.writeHead(404).end();
      return;
    }
    const chunks: Buffer[] = [];
    for await (const c of req) chunks.push(c as Buffer);
    const server = buildServer();
    const transport = new StreamableHTTPServerTransport({ sessionIdGenerator: undefined, enableJsonResponse: true });
    res.on(class="s">"close", () => { transport.close(); server.close(); });
    await server.connect(transport);
    await transport.handleRequest(req, res, JSON.parse(Buffer.concat(chunks).toString(class="s">"utf8")));
  });
  http.listen(port, class="s">"127.0.0.1", () => console.error(class="s">`${SERVER_NAME} listening on http://127.0.0.1:${port}/mcp`));
}

import type { ApiResult } from class="s">"./types.js";

export type ToolResponse = {
  content: { type: class="s">"text"; text: string }[];
  structuredContent?: Record<string, unknown>;
  isError?: boolean;
  [k: string]: unknown;
};

export function errorResponse(message: string): ToolResponse {
  return { isError: true, content: [{ type: class="s">"text", text: class="s">`Error: ${message}` }] };
}

/** Turns an API result into a tool response: structuredContent + text in the requested format. */
export function respond(
  result: ApiResult<unknown>,
  build: (body: any) => { data: Record<string, unknown>; markdown: string },
  format: ResponseFormat
): ToolResponse {
  if (!result.ok) return errorResponse(class="s">`${result.error} (HTTP ${result.status})`);
  const { data, markdown } = build(result.body);
  let text = format === ResponseFormat.JSON ? JSON.stringify(data, null, 2) : markdown;
  if (text.length > CHARACTER_LIMIT) {
    text = text.slice(0, CHARACTER_LIMIT) + class="s">`\n\n[Truncated at ${CHARACTER_LIMIT} characters. Use 'limit'/'offset' or filters to narrow the result.]`;
  }
  return { content: [{ type: class="s">"text", text }], structuredContent: data };
}

// GET /appointments
export async function listAppointments(q: { date?: string; therapistId?: string; status?: AppointmentStatus }) {
  const rows = appointments
    .filter((a) => !q.date || a.start.startsWith(q.date))
    .filter((a) => !q.therapistId || a.therapistId === q.therapistId)
    .filter((a) => !q.status || a.status === q.status)
    .sort((a, b) => a.start.localeCompare(b.start));
  return ok(rows);
}

// POST /appointments
export async function bookAppointment(b: { patientId: string; therapistId: string; start: string; durationMin?: number }) {
  if (!patients.some((p) => p.id === b.patientId)) return fail(404, class="s">`Patient '${b.patientId}' not found. Use tidewater_search_patients to find a valid patient ID.`);
  if (!therapists.some((t) => t.id === b.therapistId)) return fail(404, class="s">`Therapist '${b.therapistId}' not found. Use tidewater_list_therapists to see valid IDs.`);
  const startMs = Date.parse(b.start);
  if (Number.isNaN(startMs)) return fail(400, class="s">`'${b.start}' is not a valid ISO 8601 date-time. Example: 2026-10-26T11:00:00Z.`);
  const durationMin = b.durationMin ?? DEFAULT_DURATION_MIN;
  const d = new Date(startMs);
  const dayStart = Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate(), DAY_START_HOUR);
  const dayEnd = Date.UTC(d.getUTCFullYear(), d.getUTCMonth(), d.getUTCDate(), DAY_END_HOUR);
  const endAt = startMs + durationMin * 60_000;
  if (startMs < dayStart || endAt > dayEnd) {
    return fail(422, class="s">`The clinic is open ${DAY_START_HOUR}:00-${DAY_END_HOUR}:00 UTC. Choose a start/duration that ends by ${DAY_END_HOUR}:00; tidewater_get_availability lists open slots.`);
  }
  const clash = appointments.find((a) => a.status === class="s">"booked" && a.therapistId === b.therapistId && overlaps(startMs, endAt, a));
  if (clash) return fail(409, class="s">`Therapist ${b.therapistId} already has appointment ${clash.id} at ${clash.start}. Call tidewater_get_availability to pick a free slot.`);
  const created: Appointment = { id: class="s">`a${nextId++}`, patientId: b.patientId, therapistId: b.therapistId, start: new Date(startMs).toISOString(), durationMin, status: class="s">"booked" };
  appointments.push(created);
  return ok(created, 201);
}

// POST /appointments/{id}/cancel
export async function cancelAppointment(id: string, b: { reason?: string; notifyPatient?: boolean }) {
  const a = appointments.find((x) => x.id === id);
  if (!a) return fail(404, class="s">`Appointment '${id}' not found. Use tidewater_list_appointments to find valid IDs.`);
  if (a.status === class="s">"done") return fail(409, class="s">`Appointment '${id}' is already completed and cannot be cancelled.`);
  const alreadyCancelled = a.status === class="s">"cancelled";
  a.status = class="s">"cancelled";
  if (b.reason) a.cancelReason = b.reason;
  return ok({ appointment: a, patientNotified: !alreadyCancelled && (b.notifyPatient ?? true), alreadyCancelled });
}

// GET /therapists
export async function listTherapists() { return ok(therapists); }

// GET /patients/search
export async function searchPatients(q: string) {
  const needle = q.trim().toLowerCase();
  return ok(patients.filter((p) => p.name.toLowerCase().includes(needle) || p.phone.replace(/\D/g, class="s">"").includes(needle.replace(/\D/g, class="s">"") || class="s">"\u0000")));
}

// GET /availability
export async function getAvailability(therapistId: string, date: string) {
  if (!therapists.some((t) => t.id === therapistId)) return fail(404, class="s">`Therapist '${therapistId}' not found. Use tidewater_list_therapists to see valid IDs.`);
  const [y, m, d] = date.split(class="s">"-").map(Number);
  const slots: string[] = [];
  for (let h = DAY_START_HOUR; h + 1 <= DAY_END_HOUR; h++) {
    const s = Date.UTC(y, m - 1, d, h);
    const e = s + DEFAULT_DURATION_MIN * 60_000;
    const busy = appointments.some((a) => a.status === class="s">"booked" && a.therapistId === therapistId && overlaps(s, e, a));
    if (!busy) slots.push(new Date(s).toISOString());
  }
  return ok(slots);
}
