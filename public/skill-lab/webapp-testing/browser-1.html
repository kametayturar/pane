from playwright.sync_api import sync_playwright
URL = class="s">'http://localhost:7112/index.html'
requests, logs = [], []
with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page(viewport={class="s">'width': 1280, class="s">'height': 720})
    page.on(class="s">'request', lambda r: requests.append(fclass="s">'{r.method} {r.url}') if class="s">'favicon' not in r.url else None)
    page.on(class="s">'console', lambda m: logs.append(fclass="s">'[{m.type}] {m.text}'))
    page.on(class="s">'pageerror', lambda e: logs.append(fclass="s">'[pageerror] {e}'))
    page.goto(URL); page.wait_for_load_state(class="s">'networkidle')
    # 1. invalid email + short password
    page.fill(class="s">'#name', class="s">'Dana Reyes'); page.fill(class="s">'#email', class="s">'dana@clinic'); page.fill(class="s">'#pw', class="s">'short')
    page.click(class="s">'button[type=submit]')
    page.wait_for_selector(class="s">'#e-email:not(:empty)')
    print(class="s">'INVALID ->', {k: page.inner_text(fclass="s">'#e-{k}') for k in (class="s">'name', class="s">'email', class="s">'pw')}, class="s">'| success box:', page.locator(class="s">'.ok').count())
    page.screenshot(path=class="s">'out/browser-1-invalid.png')
    # 2. valid
    page.fill(class="s">'#email', class="s">'dana@reyesphysio.com'); page.fill(class="s">'#pw', class="s">'correct-horse-9'); page.select_option(class="s">'#size', label=class="s">'4–10 therapists')
    page.click(class="s">'button[type=submit]')
    page.wait_for_selector(class="s">'.ok')
    print(class="s">'VALID   ->', {k: page.inner_text(fclass="s">'#e-{k}') for k in (class="s">'name', class="s">'email', class="s">'pw')}, class="s">'| success:', page.inner_text(class="s">'.ok'))
    page.screenshot(path=class="s">'out/browser-1-valid.png')
    print(class="s">'requests after load:', requests)
    print(class="s">'console/pageerrors:', logs or class="s">'none')
    browser.close()

from playwright.sync_api import sync_playwright
URL = class="s">'http://localhost:7112/index.html'
with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page(viewport={class="s">'width': 1280, class="s">'height': 720})
    page.goto(URL)
    page.wait_for_load_state(class="s">'networkidle')
    page.screenshot(path=class="s">'out/recon-signup.png', full_page=True)
    print(class="s">'buttons:', [b.inner_text() for b in page.locator(class="s">'button').all()])
    print(class="s">'inputs:', [(i.get_attribute(class="s">'id'), i.get_attribute(class="s">'type')) for i in page.locator(class="s">'input').all()])
    browser.close()
