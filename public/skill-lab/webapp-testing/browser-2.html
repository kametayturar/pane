from playwright.sync_api import sync_playwright
URL = class="s">'http://localhost:7112/index.html#book'
requests, logs = [], []
with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page(viewport={class="s">'width': 1280, class="s">'height': 720})
    page.on(class="s">'request', lambda r: requests.append(fclass="s">'{r.method} {r.url}') if class="s">'favicon' not in r.url else None)
    page.on(class="s">'console', lambda m: logs.append(fclass="s">'[{m.type}] {m.text}'))
    page.on(class="s">'pageerror', lambda e: logs.append(fclass="s">'[pageerror] {e}'))
    page.goto(URL); page.wait_for_load_state(class="s">'networkidle')
    print(class="s">'selected slot highlighted:', page.locator(class="s">'.slot.sel').inner_text(), class="s">'| JS `selected` =', page.evaluate(class="s">'typeof selected'))
    # A: click Book with the default (highlighted) slot
    page.click(class="s">'#book'); page.wait_for_timeout(800)
    page.screenshot(path=class="s">'out/browser-2-click.png')
    print(class="s">'A) no slot clicked -> #result:', repr(page.inner_html(class="s">'#result')), class="s">'| requests:', requests[1:])
    print(class="s">'A) console:', logs)
    # B: pick the slot first, then Book
    logs.clear()
    page.click(class="s">'.slot:has-text("10:30")'); page.click(class="s">'#book'); page.wait_for_selector(class="s">'#result .ok')
    print(class="s">'B) after clicking a slot -> #result:', page.inner_text(class="s">'#result'), class="s">'| requests:', requests[1:], class="s">'| console:', logs)
    page.screenshot(path=class="s">'out/browser-2-after-slot.png')
    browser.close()

from playwright.sync_api import sync_playwright
URL = class="s">'http://localhost:7112/index.html'
with sync_playwright() as p:
    browser = p.chromium.launch(headless=True)
    page = browser.new_page(viewport={class="s">'width': 1280, class="s">'height': 720})
    page.goto(URL)
    page.wait_for_load_state(class="s">'networkidle')
    page.screenshot(path=class="s">'out/recon-signup.png', full_page=True)
    print(class="s">'buttons:', [b.inner_text() for b in page.locator(class="s">'button').all()])
    print(class="s">'inputs:', [(i.get_attribute(class="s">'id'), i.get_attribute(class="s">'type')) for i in page.locator(class="s">'input').all()])
    browser.close()
