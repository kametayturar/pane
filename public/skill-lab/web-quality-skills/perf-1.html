import * as React from class="s">"react"
import groupBy from class="s">"lodash/groupBy" // per-method import: pulls in only groupBy, not all of lodash
import type { Stats } from class="s">"./StatsChart"

// Component-based code splitting: the chart (and Chart.js) is fetched only when there are stats to draw.
const StatsChart = React.lazy(() => import(class="s">"./StatsChart"))

type Appointment = { id: string; patient: string; start: string; therapistId: string; status: class="s">"booked" | class="s">"cancelled" | class="s">"done" }
type Therapist = { id: string; name: string }

// moment (~60 KB gzip-ish of JS for two format() calls) replaced by the built-in Intl formatter, created once.
const timeFmt = new Intl.DateTimeFormat(class="s">"en-GB", { hour: class="s">"2-digit", minute: class="s">"2-digit", hourCycle: class="s">"h23" })
const clockFmt = new Intl.DateTimeFormat(class="s">"en-GB", { hour: class="s">"2-digit", minute: class="s">"2-digit", second: class="s">"2-digit", hourCycle: class="s">"h23" })

// Debounce expensive operations: filtering and re-rendering 1500 rows waits until typing pauses.
function useDebounced<T>(value: T, ms: number): T {
  const [v, setV] = React.useState(value)
  React.useEffect(() => {
    const id = setTimeout(() => setV(value), ms)
    return () => clearTimeout(id)
  }, [value, ms])
  return v
}

const getJson = <T,>(url: string, signal: AbortSignal): Promise<T> => fetch(url, { signal }).then((r) => r.json())

export default function Dashboard({ clinicId }: { clinicId: string }) {
  const [appointments, setAppointments] = React.useState<Appointment[]>([])
  const [therapists, setTherapists] = React.useState<Therapist[]>([])
  const [stats, setStats] = React.useState<Stats | null>(null)
  const [search, setSearch] = React.useState(class="s">"")
  const query = useDebounced(search, 150)

  // The request chart showed three sequential round trips; they are independent, so issue them together.
  React.useEffect(() => {
    const ctrl = new AbortController()
    const base = class="s">`/api/clinics/${clinicId}`
    Promise.all([
      getJson<Appointment[]>(class="s">`${base}/appointments`, ctrl.signal),
      getJson<Therapist[]>(class="s">`${base}/therapists`, ctrl.signal),
      getJson<Stats>(class="s">`${base}/stats`, ctrl.signal),
    ])
      .then(([a, t, s]) => {
        setAppointments(a)
        setTherapists(t)
        setStats(s)
      })
      .catch((e) => {
        if (e.name !== class="s">"AbortError") throw e
      })
    return () => ctrl.abort()
  }, [clinicId])

  // Sort only when the data changes; filter+group only when the data or the debounced query changes.
  const byTherapist = React.useMemo(() => {
    const q = query.toLowerCase()
    const sorted = appointments
      .map((a) => ({ a, ts: Date.parse(a.start) }))
      .sort((x, y) => x.ts - y.ts)
      .map((x) => x.a)
    return groupBy(
      sorted.filter((a) => a.patient.toLowerCase().includes(q)),
      class="s">"therapistId",
    )
  }, [appointments, query])

  const cancel = React.useCallback((id: string) => setAppointments((curr) => curr.filter((x) => x.id !== id)), [])

  return (
    <div>
      <h1>
        Today · <Clock />
      </h1>
      <input value={search} onChange={(e) => setSearch(e.target.value)} placeholder=class="s">"Search patients" />
      {stats && (
        <React.Suspense fallback={null}>
          <StatsChart data={stats} />
        </React.Suspense>
      )}
      {therapists.map((t) => (
        // Long list: let the browser skip layout/paint of off-screen therapist sections.
        <section key={t.id} style={{ contentVisibility: class="s">"auto", containIntrinsicSize: class="s">"auto 600px" }}>
          <h2>{t.name}</h2>
          {(byTherapist[t.id] || []).map((a) => (
            <Row key={a.id} appointment={a} therapistName={t.name} onCancel={cancel} />
          ))}
        </section>
      ))}
    </div>
  )
}

// The once-a-second tick re-rendered the whole dashboard (long task every second in the trace); isolate it.
function Clock() {
  const [now, setNow] = React.useState(() => Date.now())
  React.useEffect(() => {
    const id = setInterval(() => setNow(Date.now()), 1000)
    return () => clearInterval(id)
  }, [])
  return <span>{clockFmt.format(now)}</span>
}

const Row = React.memo(function Row({ appointment, therapistName, onCancel }: { appointment: Appointment; therapistName: string; onCancel: (id: string) => void }) {
  return (
    <div style={{ display: class="s">"flex", gap: 8 }}>
      <span>{timeFmt.format(Date.parse(appointment.start))}</span>
      <span>{appointment.patient}</span>
      <span>{therapistName}</span>
      <button onClick={() => onCancel(appointment.id)}>Cancel</button>
    </div>
  )
})

import * as React from class="s">"react"
// Tree-shaking: register only what a line chart needs instead of "chart.js/auto" (which pulls in every chart type).
import { Chart, LineController, LineElement, PointElement, LinearScale, CategoryScale } from class="s">"chart.js"

Chart.register(LineController, LineElement, PointElement, LinearScale, CategoryScale)

export type Stats = { labels: string[]; values: number[] }

export default function StatsChart({ data }: { data: Stats }) {
  const ref = React.useRef<HTMLCanvasElement>(null)
  React.useEffect(() => {
    const chart = new Chart(ref.current!, { type: class="s">"line", data: { labels: data.labels, datasets: [{ data: data.values }] } })
    return () => chart.destroy() // the old effect created a new chart on every render and never cleaned up
  }, [data])
  return <canvas ref={ref} />
}
