playwright-test-signup.js
const path = require(class="s">'node:path');
const helpers = require(class="s">`${process.env.PW_SKILL_DIR}/lib/helpers`);
const targetUrl = process.env.TARGET_URL || class="s">'http://127.0.0.1:7131';
const art = process.env.PW_ARTIFACT_DIR;

(async () => {
  const browser = await helpers.launchBrowser(class="s">'chromium');
  try {
    const context = await helpers.createContext(browser);
    const page = await context.newPage();
    const reqs = [];
    page.on(class="s">'request', r => { if (!r.url().endsWith(class="s">'.html') && r.url() !== targetUrl + class="s">'/') reqs.push(r.method() + class="s">' ' + r.url()); });
    page.on(class="s">'pageerror', e => console.log(class="s">'PAGEERROR:', e.message));
    await page.goto(targetUrl + class="s">'/#signup');
    await page.getByRole(class="s">'heading', { name: class="s">'Create your clinic account' }).waitFor();

    // Case 1: invalid email
    await page.getByLabel(class="s">'Full name').fill(class="s">'Dana Okafor');
    await page.getByLabel(class="s">'Work email').fill(class="s">'dana@clinic');
    await page.getByLabel(class="s">'Password').fill(class="s">'hunter2hunter2');
    await page.getByRole(class="s">'button', { name: class="s">'Create account' }).click();
    const emailErr = await page.locator(class="s">'#e-email').textContent();
    const status1 = await page.getByRole(class="s">'status').count();
    console.log(class="s">'CASE1 email error text:', JSON.stringify(emailErr));
    console.log(class="s">'CASE1 other errors:', JSON.stringify([await page.locator(class="s">'#e-name').textContent(), await page.locator(class="s">'#e-pw').textContent()]));
    console.log(class="s">'CASE1 success banner count:', status1);
    await page.screenshot({ path: path.join(art, class="s">'browser-1-invalid.png') });

    // Case 2: valid data
    await page.getByLabel(class="s">'Work email').fill(class="s">'dana@clinic.example');
    await page.getByLabel(class="s">'Clinic size').selectOption({ label: class="s">'4–10 therapists' });
    await page.getByRole(class="s">'button', { name: class="s">'Create account' }).click();
    await page.getByRole(class="s">'status').waitFor();
    console.log(class="s">'CASE2 email error text:', JSON.stringify(await page.locator(class="s">'#e-email').textContent()));
    console.log(class="s">'CASE2 status banner:', JSON.stringify(await page.getByRole(class="s">'status').textContent()));
    console.log(class="s">'CASE2 network requests during flow:', JSON.stringify(reqs));
    await page.screenshot({ path: path.join(art, class="s">'browser-1-valid.png') });
    // extra: empty password short path
    await page.reload();
    await page.getByLabel(class="s">'Full name').fill(class="s">'A');
    await page.getByLabel(class="s">'Work email').fill(class="s">'a@b.co');
    await page.getByLabel(class="s">'Password').fill(class="s">'short');
    await page.getByRole(class="s">'button', { name: class="s">'Create account' }).click();
    console.log(class="s">'CASE3 short password error:', JSON.stringify(await page.locator(class="s">'#e-pw').textContent()), class="s">'banner:', await page.getByRole(class="s">'status').count());
  } finally { await browser.close(); }
})();
