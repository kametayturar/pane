import sys, subprocess, glob, os
sys.path.insert(0, class="s">"skill")
from PIL import Image
from core.gif_builder import GIFBuilder
from core.validators import validate_gif, is_slack_ready

SRC = class="s">"/Users/kametayturar/agentic-ui/ai-answer/docs/skill-lab/fixtures/video/clip.mp4"
W, H, FPS, COLORS = int(sys.argv[1]), int(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4])
out = sys.argv[5]
os.makedirs(class="s">"frames", exist_ok=True)
for f in glob.glob(class="s">"frames/*.png"): os.remove(f)
subprocess.run([class="s">"ffmpeg",class="s">"-v",class="s">"error",class="s">"-y",class="s">"-i",SRC,class="s">"-vf",fclass="s">"fps={FPS},scale={W}:{H}:flags=lanczos",class="s">"frames/f%03d.png"],check=True)
builder = GIFBuilder(width=W, height=H, fps=FPS)
for p in sorted(glob.glob(class="s">"frames/*.png")):
    builder.add_frame(Image.open(p))
info = builder.save(out, num_colors=COLORS, remove_duplicates=False)
print(validate_gif(out, is_emoji=False)[0], class="s">"slack_ready:", is_slack_ready(out))
