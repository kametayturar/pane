import json, math, sys, pathlib

W = H = 240
FR, OP = 60, 120                 # 2 s loop
N, GAP, D = 5, 40, 24            # dots, spacing (px), diameter (px)
CY, AMP = 120, 26                # baseline y and wave amplitude (px)
PERIOD = OP                      # one wave period == loop length -> frame 0 == frame OP
STEP = PERIOD // N               # phase offset between neighbouring dots (24 frames)
ACCENT = [0.0, 0.62, 0.62, 1.0]  # teal, exposed as slot accentColor
EASE = {class="s">"o": {class="s">"x": [0.45], class="s">"y": [0]}, class="s">"i": {class="s">"x": [0.55], class="s">"y": [1]}}  # soft S-curve (sine-like) for a continuous loop

def kf(t, s):
    return {class="s">"t": t, **EASE, class="s">"s": s}

def dot(k):
    x = W / 2 + (k - (N - 1) / 2) * GAP
    phase = k * STEP
    # extrema every PERIOD/4 .. alternate crest (up) / trough (down); run beyond [0, OP] so the loop is seamless
    p, sc, op = [], [], []
    for n in range(-4, 10):
        t = phase + PERIOD // 4 + n * (PERIOD // 2)   # first crest at phase+30
        up = (n % 2 == 0)
        p.append(kf(t, [x, CY - AMP if up else CY + AMP, 0]))
        sc.append(kf(t, [118, 118, 100] if up else [78, 78, 100]))
        op.append(kf(t, [100] if up else [50]))
